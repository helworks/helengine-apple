cmake_minimum_required(VERSION 3.31)
project(HelEngineApple VERSION 0.1.0 LANGUAGES C CXX OBJC OBJCXX)
if(NOT APPLE)
    message(FATAL_ERROR "HelEngine Apple requires the actual macOS/iOS SDK; configure on the approved Mac")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_OBJCXX_STANDARD 20)
set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
set(HELENGINE_NATIVE_PROFILE "Debug" CACHE STRING "Debug, Release or Profiler")
set_property(CACHE HELENGINE_NATIVE_PROFILE PROPERTY STRINGS Debug Release Profiler)
set(HELENGINE_CORE_CPP_ROOT "" CACHE PATH "Generated engine C++ root")
set(HELENGINE_CODE_ROOT "" CACHE PATH "Generated project C++ root")
set(HELENGINE_CONTENT_ROOT "" CACHE PATH "Cooked project content root")
option(HELENGINE_DIAGNOSTIC_ONLY "Explicitly build the native diagnostic, without claiming a generated core player" OFF)
if(NOT HELENGINE_NATIVE_PROFILE MATCHES "^(Debug|Release|Profiler)$")
    message(FATAL_ERROR "Unknown HELENGINE_NATIVE_PROFILE: ${HELENGINE_NATIVE_PROFILE}")
endif()
if(HELENGINE_NATIVE_PROFILE STREQUAL "Profiler")
    message(FATAL_ERROR "Profiler is not validated: Tracy generated-scope and capture integration is required")
endif()
if(NOT HELENGINE_DIAGNOSTIC_ONLY)
    foreach(root HELENGINE_CORE_CPP_ROOT HELENGINE_CODE_ROOT HELENGINE_CONTENT_ROOT)
        if(NOT IS_DIRECTORY "${${root}}")
            message(FATAL_ERROR "${root} must identify the real generated/cooked directory; diagnostic requires explicit HELENGINE_DIAGNOSTIC_ONLY=ON")
        endif()
    endforeach()
    message(FATAL_ERROR "Generated core player integration is not complete; diagnostic is available explicitly and is not a DemoDisc player")
endif()
if(NOT CMAKE_OSX_ARCHITECTURES)
    set(CMAKE_OSX_ARCHITECTURES arm64 CACHE STRING "Validated Apple architecture" FORCE)
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
    set(HELENGINE_TARGET ios)
else()
    set(HELENGINE_TARGET macos)
endif()
include(FetchContent)
set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
set(SDL_EXAMPLES OFF CACHE BOOL "" FORCE)
FetchContent_Declare(SDL3 URL https://www.libsdl.org/release/SDL3-3.2.28.tar.gz URL_HASH SHA256=1330671214d146f8aeb1ed399fc3e081873cdb38b5189d1f8bb6ab15bbc04211 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(SDL3)
find_program(XCRUN xcrun REQUIRED)
if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
    if(CMAKE_OSX_SYSROOT MATCHES "[Ss]imulator")
        set(HELENGINE_METAL_SDK iphonesimulator)
    else()
        set(HELENGINE_METAL_SDK iphoneos)
    endif()
else()
    set(HELENGINE_METAL_SDK macosx)
endif()
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/helengine.metallib"
    COMMAND "${XCRUN}" --sdk "${HELENGINE_METAL_SDK}" metal -c "${CMAKE_CURRENT_SOURCE_DIR}/src/shaders/primitives.metal" -o "${CMAKE_CURRENT_BINARY_DIR}/primitives.air"
    COMMAND "${XCRUN}" --sdk "${HELENGINE_METAL_SDK}" metallib "${CMAKE_CURRENT_BINARY_DIR}/primitives.air" -o "${CMAKE_CURRENT_BINARY_DIR}/helengine.metallib"
    DEPENDS src/shaders/primitives.metal VERBATIM)
add_custom_target(helengine_metal_library ALL DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/helengine.metallib")
add_library(helengine_metal STATIC src/apple/metal_renderer.mm)
target_include_directories(helengine_metal PUBLIC src)
target_compile_options(helengine_metal PRIVATE -fobjc-arc -Wall -Wextra -Werror)
target_link_libraries(helengine_metal PUBLIC "-framework Metal" "-framework QuartzCore" "-framework Foundation")
if(IS_DIRECTORY "${HELENGINE_CORE_CPP_ROOT}")
    add_library(helengine_core_2d_bridge OBJECT src/apple/core_render_manager_2d.cpp)
    target_include_directories(helengine_core_2d_bridge PRIVATE src "${HELENGINE_CORE_CPP_ROOT}")
    target_compile_options(helengine_core_2d_bridge PRIVATE -Wall -Wextra)
endif()
add_executable(helengine_${HELENGINE_TARGET} MACOSX_BUNDLE src/apple/diagnostic_application.mm)
target_link_libraries(helengine_${HELENGINE_TARGET} PRIVATE helengine_metal SDL3::SDL3-static)
target_compile_options(helengine_${HELENGINE_TARGET} PRIVATE -fobjc-arc -Wall -Wextra -Werror)
set_target_properties(helengine_${HELENGINE_TARGET} PROPERTIES MACOSX_BUNDLE_GUI_IDENTIFIER org.helworks.helengine.diagnostic MACOSX_BUNDLE_BUNDLE_NAME HelEngineAppleDiagnostic XCODE_ATTRIBUTE_PRODUCT_BUNDLE_IDENTIFIER org.helworks.helengine.diagnostic XCODE_ATTRIBUTE_CODE_SIGNING_ALLOWED NO)
set_source_files_properties("${CMAKE_CURRENT_BINARY_DIR}/helengine.metallib" PROPERTIES MACOSX_PACKAGE_LOCATION Resources GENERATED TRUE)
target_sources(helengine_${HELENGINE_TARGET} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/helengine.metallib")
add_dependencies(helengine_${HELENGINE_TARGET} helengine_metal_library)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
    enable_testing()
    add_executable(frame_lifecycle_tests tests/frame_lifecycle_tests.cpp)
    target_include_directories(frame_lifecycle_tests PRIVATE src)
    add_test(NAME frame_lifecycle COMMAND frame_lifecycle_tests)
    add_executable(metal_readback_tests tests/metal_readback_tests.mm)
    target_link_libraries(metal_readback_tests PRIVATE helengine_metal)
    target_compile_options(metal_readback_tests PRIVATE -fobjc-arc)
    add_dependencies(metal_readback_tests helengine_metal_library)
    add_test(NAME metal_readback COMMAND metal_readback_tests)
    add_executable(metal_authored_library_tests tests/metal_authored_library_tests.mm)
    target_compile_options(metal_authored_library_tests PRIVATE -fobjc-arc -Wall -Wextra -Werror)
    target_link_libraries(metal_authored_library_tests PRIVATE "-framework Metal" "-framework Foundation")
endif()
